rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function esAdmin() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email in [
          'admin.email',
          'admin.email',
          'admin.email'
        ];
    }

    function esUser() {
      return request.auth != null
        && request.auth.token.email_verified == true;
    }

    // Los dos administradores principales (los dos primeros correos).
    // Al publicar las reglas, pon aquí sus dos correos.
    function esAdminPrincipal() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email in [
          'primer.admin@email',
          'segundo.admin@email'
        ];
    }

    function votoValido() {
      return request.resource.data.votos == 1
        && request.resource.data.suma >= 1
        && request.resource.data.suma <= 5;
    }

    function sumaVoto() {
      return request.resource.data.votos == resource.data.votos + 1
        && request.resource.data.suma - resource.data.suma >= 1
        && request.resource.data.suma - resource.data.suma <= 5;
    }

    function solicitudValida() {
      let d = request.resource.data;
      return d.keys().hasOnly(['titulo', 'autor', 'mensaje'])
        && d.titulo is string && d.titulo.size() <= 200
        && d.autor is string && d.autor.size() > 0 && d.autor.size() <= 100
        && d.mensaje is string && d.mensaje.size() <= 3000;
    }

    match /poemas/{poemaId} {
      allow read: if true;
      allow create: if esAdmin() || votoValido();
      allow update: if esAdmin() || sumaVoto();
      allow delete: if esAdmin();

      allow create, update, delete: if esAdmin();
    }

    match /poemas-publicados/{id} {
      allow read: if true;
      allow create, update, delete: if esAdmin();
    }

    match /solicitudes/{id} {
      allow create: if esUser() && solicitudValida();
      allow read, update, delete: if esAdmin();
    }

    // Autores: los lee todo el mundo (la página principal muestra sus nombres
    // y colores). Editar puede cualquier admin; agregar y eliminar solo los
    // dos administradores principales.
    match /autores/{id} {
      allow read: if true;
      allow update: if esAdmin();
      allow create, delete: if esAdminPrincipal();
    }

    // Borradores: solo los admins los ven y los usan.
    match /borradores/{id} {
      allow read, create, update, delete: if esAdmin();
    }

    // Visitas: cualquier visitante suma 1 al abrir un poema; solo los admins
    // pueden leer el conteo. La suma solo puede crecer de 1 en 1 y solo se
    // pueden escribir los campos de la visita.
    match /visitas/{id} {
      allow read: if esAdmin();
      allow create: if request.resource.data.keys().hasOnly(['visitas', 'titulo', 'autor', 'fecha'])
        && request.resource.data.visitas == 1
        && request.resource.data.titulo is string
        && request.resource.data.autor is string;
      allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(['visitas', 'titulo', 'autor', 'fecha'])
        && request.resource.data.visitas == resource.data.visitas + 1;
    }
  }
}
